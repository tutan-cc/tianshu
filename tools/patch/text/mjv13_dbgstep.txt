  function dbgStep() {
    if (!G.on || G.finished || G.result) return "off";
    var E = G.E;
    if (!E) return "off";
    if (E.phase === "over") return "over";
    var r = E.aiStep();
    playNewSfx(); updateHud();
    return r;
  }
